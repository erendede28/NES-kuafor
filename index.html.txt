    <script>
        document.getElementById('bDate').min = new Date().toISOString().split('T')[0];
        let currentUser = localStorage.getItem('nes_logged_user') || null;

        function updateNavAuthUI() {
            const authButtons = document.getElementById('authButtonsContainer');
            const userProfile = document.getElementById('userProfileContainer');
            const navUserName = document.getElementById('navUserName');

            if (currentUser) {
                authButtons.classList.add('hidden');
                userProfile.classList.remove('hidden');
                userProfile.classList.add('flex');
                navUserName.innerText = currentUser;
                document.getElementById('bName').value = currentUser;
                document.getElementById('bName').disabled = true;
            } else {
                authButtons.classList.remove('hidden');
                userProfile.classList.remove('flex');
                userProfile.classList.add('hidden');
                document.getElementById('bName').value = '';
                document.getElementById('bName').disabled = false;
            }
        }
        updateNavAuthUI();

        function checkUserLogin() {
            return currentUser !== null;
        }

        function getApps() {
            return JSON.parse(localStorage.getItem('nes_apps')) || [];
        }

        function saveApps(apps) {
            localStorage.setItem('nes_apps', JSON.stringify(apps));
        }

        function handleBooking(e) {
            e.preventDefault();
            const warningEl = document.getElementById('bookingWarning');
            const successEl = document.getElementById('bookingSuccess');

            if (!checkUserLogin()) {
                warningEl.classList.remove('hidden');
                successEl.classList.add('hidden');
                openLoginModal();
                return;
            }

            warningEl.classList.add('hidden');
            const fileInput = document.getElementById('bFile');

            const saveRecord = (img64) => {
                const apps = getApps();
                apps.push({
                    id: Date.now(),
                    name: currentUser,
                    phone: document.getElementById('bPhone').value,
                    service: document.getElementById('bService').value,
                    date: document.getElementById('bDate').value,
                    time: document.getElementById('bTime').value,
                    note: document.getElementById('bNote').value,
                    image: img64,
                    status: 'Beklemede',
                    reply: ''
                });
                saveApps(apps);
                e.target.reset();
                document.getElementById('bName').value = currentUser;
                document.getElementById('bName').disabled = true;
                successEl.classList.remove('hidden');
                setTimeout(() => successEl.classList.add('hidden'), 4000);
            };

            if (fileInput.files && fileInput.files[0]) {
                const reader = new FileReader();
                reader.onload = function(uploadEvent) {
                    saveRecord(uploadEvent.target.result);
                };
                reader.readAsDataURL(fileInput.files[0]);
            } else {
                saveRecord('');
            }
        }

        // Admin
        function openAdminModal() { document.getElementById('adminModal').classList.remove('hidden'); }
        function closeAdminModal() { document.getElementById('adminModal').classList.add('hidden'); }

        function verifyAdmin() {
            const val = document.getElementById('adminPassInput').value;
            if(val === '1234') {
                document.getElementById('adminAuthScreen').classList.add('hidden');
                document.getElementById('adminPanelContent').classList.remove('hidden');
                renderAdminTable();
            } else {
                document.getElementById('adminErr').classList.remove('hidden');
            }
        }

        function renderAdminTable() {
            const apps = getApps();
            const tbody = document.getElementById('adminTableBody');
            tbody.innerHTML = '';
            if(apps.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" class="text-center p-6 text-gray-500">Henüz randevu yok.</td></tr>';
                return;
            }
            apps.forEach(app => {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-gray-900/50";
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${app.name}<br><span class="text-xs text-gray-400">${app.phone}</span></td>
                    <td class="p-3">
                        <span class="gold-text">${app.service}</span><br>
                        <span class="text-xs text-gray-400">${app.note || 'Not yok'}</span>
                        ${app.image ? `<br><button onclick="openLightboxImage('${app.image}', '${app.note}', ${app.id})" class="text-xs text-[#D4AF37] underline mt-1 inline-block"><i class="fa-solid fa-image"></i> Görseli Gör</button>` : ''}
                    </td>
                    <td class="p-3 text-gray-300">${app.date} <br><b>${app.time}</b></td>
                    <td class="p-3"><span class="px-2.5 py-1 rounded-full text-xs font-medium ${app.status==='Onaylandı'?'bg-emerald-900/50 text-emerald-400 border border-emerald-800':'bg-amber-900/50 text-amber-400 border border-amber-800'}">${app.status}</span></td>
                    <td class="p-3 space-x-2">
                        <button onclick="updateStatus(${app.id}, 'Onaylandı')" class="text-emerald-400 hover:text-emerald-300" title="Onayla"><i class="fa-solid fa-check"></i></button>
                        <button onclick="updateStatus(${app.id}, 'İptal')" class="text-red-400 hover:text-red-300" title="İptal Et"><i class="fa-solid fa-ban"></i></button>
                        <button onclick="openLightboxImage('${app.image || ''}', '${app.note || 'Not yok'}', ${app.id})" class="text-blue-400 hover:text-blue-300" title="Mesaj & Yanıt"><i class="fa-solid fa-message"></i></button>
                        <button onclick="deleteApp(${app.id})" class="text-gray-500 hover:text-red-400" title="Sil"><i class="fa-solid fa-trash"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        let activeLightboxId = null;
        function openLightboxImage(imgSrc, noteText, appId) {
            activeLightboxId = appId;
            const container = document.getElementById('lightboxMediaContainer');
            if(imgSrc) {
                container.innerHTML = `<img src="${imgSrc}" class="w-full h-auto rounded-xl border border-gray-800">`;
            } else {
                container.innerHTML = `<div class="text-gray-500 text-sm italic p-8 text-center bg-[#0A0A0A] rounded-xl">Fotoğraf yüklenmemiş</div>`;
            }
            document.getElementById('lightboxNoteText').innerText = noteText;
            document.getElementById('adminReplyText').value = '';
            document.getElementById('lightboxModal').classList.remove('hidden');
        }

        function closeLightbox() { document.getElementById('lightboxModal').classList.add('hidden'); }

        function sendAdminReply() {
            const replyVal = document.getElementById('adminReplyText').value;
            if(!replyVal) return alert('Lütfen yanıt yazın.');
            const apps = getApps();
            const target = apps.find(a => a.id === activeLightboxId);
            if(target) {
                target.reply = replyVal;
                saveApps(apps);
                alert('Yanıt başarıyla müşteriye iletildi!');
                closeLightbox();
                renderAdminTable();
            }
        }

        function updateStatus(id, st) {
            const apps = getApps();
            const target = apps.find(a => a.id === id);
            if(target) { target.status = st; saveApps(apps); renderAdminTable(); }
        }

        function deleteApp(id) {
            let apps = getApps();
            apps = apps.filter(a => a.id !== id);
            saveApps(apps);
            renderAdminTable();
        }

        function clearAllApp() {
            if(confirm('Tüm kayıtlar silinsin mi?')) {
                localStorage.removeItem('nes_apps');
                renderAdminTable();
            }
        }

        function closeLoginModal() { document.getElementById('loginModal').classList.add('hidden'); }
        function openRegisterModal() { document.getElementById('registerModal').classList.remove('hidden'); }
        function closeRegisterModal() { document.getElementById('registerModal').classList.add('hidden'); }

        function handleRegisterSubmit() {
            const name = document.getElementById('regName').value;
            const pass = document.getElementById('regPass').value;
            const err = document.getElementById('regErrorMsg');

            if(!name || pass.length < 6 || !/[A-Z]/.test(pass)) {
                err.innerText = "Şifre en az 6 karakter olmalı ve 1 büyük harf içermelidir!";
                err.classList.remove('hidden');
                return;
            }
            localStorage.setItem('nes_reg_name', name);
            localStorage.setItem('nes_reg_pass', pass);
            alert('Kayıt başarılı! Şimdi giriş yapabilirsiniz.');
            closeRegisterModal();
            openLoginModal();
        }

        function handleLoginSubmit() {
            const name = document.getElementById('loginName').value;
            const pass = document.getElementById('loginPass').value;
            const err = document.getElementById('loginErrorMsg');
            const savedName = localStorage.getItem('nes_reg_name');
            const savedPass = localStorage.getItem('nes_reg_pass');

            if(savedName && name === savedName && pass === savedPass) {
                currentUser = name;
                localStorage.setItem('nes_logged_user', name);
                closeLoginModal();
                updateNavAuthUI();
                alert('Giriş başarılı! Artık randevu oluşturabilirsiniz.');
            } else {
                err.innerText = "Hatalı Ad Soyad veya Şifre!";
                err.classList.remove('hidden');
            }
        }

        function handleLogout() {
            currentUser = null;
            localStorage.removeItem('nes_logged_user');
            updateNavAuthUI();
            alert('Çıkış yapıldı.');
        }
